```rust
/// A file is "text" if its first 8KB contain no NUL byte.
/// Unreadable/missing files are not text.
pub fn is_text_file(path: &Path) -> bool {
    let Ok(mut f) = File::open(path) else { return false };
    let mut buf = [0u8; 8192];
    let Ok(n) = f.read(&mut buf) else { return false };
    !buf[..n].contains(&0)
}
```

```rust
use std::io::{self, Write};

/// Write via a temp file in the same directory + rename, so a crash
/// mid-write can never truncate the destination.
pub fn atomic_write(path: &Path, contents: &[u8]) -> io::Result<()> {
    let dir = path.parent().unwrap_or(Path::new("."));
    let name = path
        .file_name()
        .ok_or_else(|| io::Error::new(io::ErrorKind::InvalidInput, "no file name"))?;
    let tmp = dir.join(format!(".{}.markdup-tmp", name.to_string_lossy()));
    {
        let mut f = File::create(&tmp)?;
        f.write_all(contents)?;
        f.sync_all()?;
    }
    std::fs::rename(&tmp, path)
}
```

```rust
use unicode_width::UnicodeWidthChar;

/// One display row: chars `[start, end)` of logical line `line`.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct VisualRow {
    pub line: usize,
    pub start: usize,
    pub end: usize,
}

pub fn ch_width(c: char) -> usize {
    if c == '\t' {
        4
    } else {
        UnicodeWidthChar::width(c).unwrap_or(0)
    }
}

pub fn layout(lines: &[String], width: usize) -> Vec<VisualRow> {
    let width = width.max(1);
    let mut rows = Vec::new();
    for (li, line) in lines.iter().enumerate() {
        let chars: Vec<char> = line.chars().collect();
        if chars.is_empty() {
            rows.push(VisualRow { line: li, start: 0, end: 0 });
            continue;
        }
        let mut start = 0;
        while start < chars.len() {
            let mut col = 0;
            let mut end = start;
            let mut last_space = None;
            while end < chars.len() {
                let w = ch_width(chars[end]);
                if col + w > width && end > start {
                    break;
                }
                if chars[end] == ' ' {
                    last_space = Some(end);
                }
                col += w;
                end += 1;
            }
            if end < chars.len() {
                // mid-line break: prefer just after the last space in this row
                if let Some(sp) = last_space {
                    if sp + 1 > start {
                        end = sp + 1;
                    }
                }
            }
            rows.push(VisualRow { line: li, start, end });
            start = end;
        }
    }
    rows
}

/// Map a logical (line, char col) cursor to (visual row index, x cells).
/// A cursor at a row's `end` belongs to the NEXT row of the same line,
/// except at the very end of the line.
pub fn cursor_position(
    rows: &[VisualRow],
    lines: &[String],
    cursor: (usize, usize),
) -> (usize, usize) {
    let (cline, ccol) = cursor;
    let mut result = 0;
    for (ri, row) in rows.iter().enumerate() {
        if row.line != cline {
            continue;
        }
        result = ri;
        let is_last_row_of_line = rows
            .get(ri + 1)
            .map_or(true, |next| next.line != cline);
        if ccol < row.end || (is_last_row_of_line && ccol >= row.end) {
            let x = lines[cline]
                .chars()
                .skip(row.start)
                .take(ccol.saturating_sub(row.start))
                .map(ch_width)
                .sum();
            return (ri, x);
        }
    }
    (result, 0)
}

/// Adjust vertical scroll (in visual rows) so `cursor_row` is visible.
pub fn scroll_top(current_top: usize, cursor_row: usize, height: usize) -> usize {
    let height = height.max(1);
    if cursor_row < current_top {
        cursor_row
    } else if cursor_row >= current_top + height {
        cursor_row + 1 - height
    } else {
        current_top
    }
}
```

**Implementation notes (read before coding):**
- Internal state: `root: PathBuf`, `expanded: HashSet<PathBuf>`, `show_hidden: bool`, `selected: usize`, `rows: Vec<Row>` — rebuilt by a private `rebuild()` that walks expanded dirs recursively. This avoids a recursive Node struct and borrow pain; directories are re-listed on each rebuild (rebuilds happen only on user actions, not per frame).
- Listing one directory:

```rust
fn list_dir(dir: &Path, show_hidden: bool) -> Vec<(PathBuf, bool)> {
    let mut out = Vec::new();
    let walker = ignore::WalkBuilder::new(dir)
        .max_depth(Some(1))
        .hidden(!show_hidden)
        .require_git(false)
        .build();
    for entry in walker.flatten() {
        if entry.depth() == 0 {
            continue; // the dir itself
        }
        let is_dir = entry.file_type().is_some_and(|t| t.is_dir());
        let path = entry.into_path();
        if is_dir || crate::fsutil::is_text_file(&path) {
            out.push((path, is_dir));
        }
    }
    out.sort_by(|a, b| b.1.cmp(&a.1).then_with(|| {
        a.0.file_name().unwrap_or_default().to_ascii_lowercase()
            .cmp(&b.0.file_name().unwrap_or_default().to_ascii_lowercase())
    }));
    out
}
```

- `rebuild()` recurses: for each `(path, is_dir)` push a `Row`; if `is_dir && expanded.contains(&path)`, recurse at depth+1.
- `.gitignore'd` dirs (e.g. `target/`) are excluded by the walker automatically. The `.git` dir is hidden, so the default hidden filter drops it; when `show_hidden` is on, skip any entry named `.git` explicitly.
- `collapse` on an expanded dir removes it (and any descendants) from `expanded`; on anything else it moves selection to the parent row if the parent is in the tree.

```rust
use ratatui_textarea::TextArea;
use std::fs;
use std::io;
use std::path::{Path, PathBuf};
use std::time::SystemTime;

use crate::fsutil;

pub enum SaveOutcome {
    Saved,
    Clean,
    Conflict,
    NoFile,
}

pub struct Editor {
    pub textarea: TextArea<'static>,
    pub path: Option<PathBuf>,
    pub dirty: bool,
    mtime: Option<SystemTime>,
}

impl Editor {
    pub fn new() -> Self {
        Editor { textarea: TextArea::default(), path: None, dirty: false, mtime: None }
    }

    pub fn open(&mut self, path: &Path) -> io::Result<()> {
        let text = fs::read_to_string(path)?;
        self.textarea = TextArea::from(text.lines().map(String::from).collect::<Vec<_>>());
        self.path = Some(path.to_path_buf());
        self.dirty = false;
        self.mtime = disk_mtime(path);
        Ok(())
    }

    pub fn mark_dirty(&mut self) {
        self.dirty = true;
    }

    fn content(&self) -> String {
        let mut s = self.textarea.lines().join("\n");
        s.push('\n');
        s
    }

    pub fn save(&mut self, force: bool) -> io::Result<SaveOutcome> {
        let Some(path) = self.path.clone() else { return Ok(SaveOutcome::NoFile) };
        if !self.dirty {
            return Ok(SaveOutcome::Clean);
        }
        if !force && self.disk_changed(&path) {
            return Ok(SaveOutcome::Conflict);
        }
        fsutil::atomic_write(&path, self.content().as_bytes())?;
        self.mtime = disk_mtime(&path);
        self.dirty = false;
        Ok(SaveOutcome::Saved)
    }

    /// Clean buffer + changed disk => reload; returns true if reloaded.
    pub fn check_external(&mut self) -> io::Result<bool> {
        let Some(path) = self.path.clone() else { return Ok(false) };
        if !self.disk_changed(&path) || self.dirty {
            return Ok(false);
        }
        self.open(&path)?;
        Ok(true)
    }

    fn disk_changed(&self, path: &Path) -> bool {
        match (self.mtime, disk_mtime(path)) {
            (Some(a), Some(b)) => a != b,
            _ => false,
        }
    }
}

fn disk_mtime(path: &Path) -> Option<SystemTime> {
    fs::metadata(path).and_then(|m| m.modified()).ok()
}
```

### Task 6: App state + key dispatch (`app.rs`)

**Files:**
- Create: `src/app.rs`
- Modify: `src/main.rs` (add `mod app;`)

**Interfaces:**
- Consumes: `Tree`, `Editor`, `SaveOutcome`
- Produces:
  - `app::Focus { Tree, Editor }`, `app::Prompt { None, NewFile(String), Search(String) }`
  - `app::App { pub tree: Tree, pub editor: Editor, pub focus: Focus, pub tree_visible: bool, pub prompt: Prompt, pub status: Option<String>, pub scroll_top: usize, pub should_quit: bool, pending_quit: bool, last_search: String, last_edit: Option<Instant> }`
  - `App::new(root: PathBuf) -> io::Result<App>`
  - `App::handle_key(&mut self, key: KeyEvent)`
  - `App::tick(&mut self)` — idle autosave (2s) + external-change check

**Key dispatch order (must be exact):**
1. If `prompt` is active: Esc cancels; Enter submits; Backspace pops; any char pushes. Nothing else.
2. Global: Ctrl+Q (quit flow), Ctrl+B (toggle `tree_visible`; if hiding while focus is Tree, focus moves to Editor).
3. Focus::Tree: `j`/`k`/arrows move; `l`/`→` expand; `h`/`←` collapse; `g`/`G` top/bottom; `Enter`/`Tab` — on a dir toggles expand, on a file: autosave current (`save(false)`; Conflict → status warning, stay), then `editor.open`, `focus = Editor`, `scroll_top = 0`; `n` → `prompt = Prompt::NewFile(String::new())`; `.` toggle hidden.
4. Focus::Editor: Esc → focus Tree. Ctrl+S → save flow. Ctrl+F → `Prompt::Search` (Enter submits and jumps; afterwards Ctrl+F re-opens prefilled with `last_search`... no: Enter in search prompt runs `search_next` and closes prompt; pressing Ctrl+F then Enter with unchanged text repeats). Everything else → `editor.textarea.input(Input::from(key))`; if it returns `true`, `editor.mark_dirty()` and `last_edit = Some(Instant::now())`. (Undo/redo Ctrl+Z/Ctrl+Y are textarea defaults and flow through `input`.)

**Save flow (Ctrl+S):** `save(false)`; on `Conflict` set status `"disk changed — Ctrl+S again to overwrite"` and set `force_next_save = true`; a Ctrl+S while `force_next_save` calls `save(true)`. Any modifying edit clears `force_next_save`.

**Quit flow (Ctrl+Q):** `save(false)`; on `Saved`/`Clean`/`NoFile` → `should_quit = true`. On `Conflict` → status `"disk changed — Ctrl+S to overwrite, Ctrl+Q again to discard"`, `pending_quit = true`; second Ctrl+Q sets `should_quit = true` without saving.

**tick():** if `dirty` and `last_edit` older than 2s → `save(false)` (Conflict → status warning, don't retry until next edit: clear `last_edit`). Then `editor.check_external()` → if reloaded, status `"reloaded (changed on disk)"`.

**New-file submit:** name is relative to the selected row's dir (selected dir itself, else the file's parent, else tree root). Reject empty or path-escaping (`..`) input with a status message. `fsutil::atomic_write(&path, b"")`, `tree.refresh()`, open the new file, focus Editor.

**search_next(query):** literal, case-sensitive, wraps around; starts one char after the cursor. Char/byte care: TextArea cursor col is a char index; `str::find` returns bytes — convert.

```rust
fn search_next(&mut self, query: &str) {
    if query.is_empty() { return; }
    let (crow, ccol) = self.editor.textarea.cursor();
    let lines: Vec<String> = self.editor.textarea.lines().to_vec();
    let n = lines.len();
    for i in 0..=n {
        let row = (crow + i) % n;
        let hay = &lines[row];
        let from_char = if i == 0 { ccol + 1 } else { 0 };
        let from_byte = hay.char_indices().nth(from_char).map(|(b, _)| b).unwrap_or(hay.len());
        if let Some(b) = hay[from_byte..].find(query) {
            let cpos = hay[..from_byte + b].chars().count();
            self.editor.textarea.move_cursor(
                ratatui_textarea::CursorMove::Jump(row as u16, cpos as u16),
            );
            self.last_search = query.to_string();
            return;
        }
    }
    self.status = Some(format!("not found: {query}"));
}
```

```rust
mod app;
mod editor;
mod fsutil;
mod tree;

use std::env;
use std::io;
use std::path::PathBuf;
use std::time::Duration;

use crossterm::event::{self, Event, KeyEventKind};

fn main() -> io::Result<()> {
    let root = match env::args().nth(1) {
        Some(p) => PathBuf::from(p),
        None => env::current_dir()?,
    };
    let mut app = app::App::new(root)?;
    let mut terminal = ratatui::init();
    let result = run(&mut terminal, &mut app);
    ratatui::restore();
    result
}

fn run(terminal: &mut ratatui::DefaultTerminal, app: &mut app::App) -> io::Result<()> {
    loop {
        terminal.draw(|f| ui::draw(f, app))?;
        if event::poll(Duration::from_millis(250))? {
            match event::read()? {
                Event::Key(k) if k.kind != KeyEventKind::Release => app.handle_key(k),
                _ => {}
            }
        } else {
            app.tick();
        }
        if app.should_quit {
            return Ok(());
        }
    }
}
```
